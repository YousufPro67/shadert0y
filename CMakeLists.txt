cmake_minimum_required(VERSION 3.13)
project(shadert0y LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(shadert0y MODULE shadert0y.cpp)
set_target_properties(shadert0y PROPERTIES PREFIX "")

if(WIN32)
    set_target_properties(shadert0y PROPERTIES SUFFIX ".dll")
    target_link_libraries(shadert0y PRIVATE opengl32 gdi32 user32)
elseif(APPLE)
    set_target_properties(shadert0y PROPERTIES SUFFIX ".so")
    find_library(OPENGL_FRAMEWORK OpenGL)
    find_library(COREFOUNDATION CoreFoundation)
    target_link_libraries(shadert0y PRIVATE ${OPENGL_FRAMEWORK} ${COREFOUNDATION})
else()
    set_target_properties(shadert0y PROPERTIES SUFFIX ".so")
    find_package(OpenGL REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(EGL REQUIRED egl)
    target_include_directories(shadert0y PRIVATE ${EGL_INCLUDE_DIRS})
    target_link_libraries(shadert0y PRIVATE OpenGL::GL ${EGL_LIBRARIES} ${CMAKE_DL_LIBS})
endif()
